class ContainerDamageDataset(Dataset):
    def __init__(self, img_dir, label_dir, classes_file=None, transform=None):
        """
        初始化集装箱破损数据集
        """
        self.img_dir = img_dir
        self.label_dir = label_dir
        self.transform = transform
        
        # 1. 获取所有图片文件名
        self.img_names = [f for f in os.listdir(img_dir) if f.endswith(('.jpg', '.png', '.jpeg'))]
        
        # 2. 解析类别文件 (用于后续可视化)
        self.classes_dict = self._parse_classes(classes_file) if classes_file else {}

    def _parse_classes(self, classes_file):
        """
        解析你提供的 data/classes.txt
        例如将 "Dent（凹陷）标记为0" 解析为字典 {0: 'Dent（凹陷）'}
        """
        classes_dict = {}
        if os.path.exists(classes_file):
            with open(classes_file, 'r', encoding='utf-8') as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    # 使用正则表达式提取文字和数字标记
                    # 匹配格式如：Dent（凹陷）标记为0
                    match = re.search(r'(.*?)\s*标记为\s*(\d+)', line)
                    if match:
                        class_name = match.group(1).strip()
                        class_id = int(match.group(2))
                        classes_dict[class_id] = class_name
        return classes_dict

    def __len__(self):
        return len(self.img_names)

    def __getitem__(self, idx):
        # ==================== 1. 加载图像 ====================
        img_name = self.img_names[idx]
        img_path = os.path.join(self.img_dir, img_name)
        
        # 使用 OpenCV 读取图像并转换为 RGB
        image = cv2.imread(img_path)
        if image is None:
            raise ValueError(f"无法读取图片，请检查路径: {img_path}")
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        height, width, _ = image.shape

        # ==================== 2. 加载并解析 YOLO 标签 ====================
        label_name = os.path.splitext(img_name)[0] + '.txt'
        label_path = os.path.join(self.label_dir, label_name)

        boxes = []
        labels = []

        if os.path.exists(label_path):
            with open(label_path, 'r', encoding='utf-8') as f:
                for line in f:
                    parts = line.strip().split()
                    if len(parts) >= 5:
                        class_id = int(parts[0])
                        cx, cy, w, h = map(float, parts[1:5])

                        # YOLO 相对坐标 -> 绝对像素坐标 (xmin, ymin, xmax, ymax)
                        xmin = (cx - w / 2) * width
                        ymin = (cy - h / 2) * height
                        xmax = (cx + w / 2) * width
                        ymax = (cy + h / 2) * height

                        boxes.append([xmin, ymin, xmax, ymax])
                        labels.append(class_id)

        # 转换为 Tensor
        boxes = torch.as_tensor(boxes, dtype=torch.float32) if boxes else torch.empty((0, 4), dtype=torch.float32)
        labels = torch.as_tensor(labels, dtype=torch.int64) if labels else torch.empty((0,), dtype=torch.int64)
        
        target = {"boxes": boxes, "labels": labels}

        # ==================== 3. 数据增强 ====================
        if self.transform:
            transformed = self.transform(image=image, bboxes=target["boxes"], class_labels=target["labels"])
            image = transformed['image']
            target["boxes"] = torch.as_tensor(transformed['bboxes'], dtype=torch.float32)

        # 基础张量化转换
        if not torch.is_tensor(image):
            image = torch.as_tensor(image.transpose((2, 0, 1)), dtype=torch.float32) / 255.0
